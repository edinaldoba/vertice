\begin{enumerate}[(1)]
\item $2x-3y-z=1$

{\it {\bf\underline{Observação}:} Perceba que a terna dada possui valores $x=1$, $y=2$ e $z=-5$ que deverão ser substituidos nas equações.}\\\vspace{-1cm}\\
