% QUESTAO

% ALTERNATIVAS

$C(x)=8+0{,}5x$ \quad e \quad R$\$\,50{,}00$
$C(x)=8$ \quad e \quad R$\$\,8{,}00$
$C(x)=0{,}5x$ \quad e \quad R$\$\,58{,}00$
$C(x)=8+x$ \quad e \quad R$\$\,100{,}00$
